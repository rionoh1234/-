jfp_dec_ryukyu_tributary = {
	is_shown = {
		jfp_is_joseon = yes
		has_journal_entry = je_jfp_ryukyu
	}
	possible = {
		is_at_war = no
		is_subject = no
		c:RYU ?= { is_subject = no }
		custom_tooltip = {
			text = jfp_tt_play_cooldown
			NOT = { has_variable = jfp_cd_ryukyu_play }
		}
	}
	when_taken = {
		set_variable = { name = jfp_cd_ryukyu_play days = 1825 }
		create_diplomatic_play = {
			target_country = c:RYU
			type = dp_make_tributary
		}
	}
	ai_chance = { value = 0 }
}

jfp_dec_formosa_expedition = {
	is_shown = {
		jfp_is_joseon = yes
		has_journal_entry = je_jfp_formosa
	}
	possible = {
		is_at_war = no
		s:STATE_FORMOSA = { any_scope_state = { NOT = { owner = root } } }
		custom_tooltip = {
			text = jfp_tt_play_cooldown
			NOT = { has_variable = jfp_cd_formosa_play }
		}
	}
	when_taken = {
		set_variable = { name = jfp_cd_formosa_play days = 1825 }
		s:STATE_FORMOSA = {
			random_scope_state = {
				limit = { NOT = { owner = root } }
				save_scope_as = jfp_target_state
				owner = { save_scope_as = jfp_target_owner }
			}
		}
		if = {
			limit = { exists = scope:jfp_target_owner }
			create_diplomatic_play = {
				target_country = scope:jfp_target_owner
				type = dp_conquer_state
				target_state = scope:jfp_target_state
			}
		}
	}
	ai_chance = { value = 0 }
}

jfp_dec_manchuria_migration = {
	is_shown = {
		jfp_is_joseon = yes
		has_journal_entry = je_jfp_manchuria_dev
		NOT = { has_variable = jfp_dec_manchuria_migration_taken }
	}
	possible = {
		custom_tooltip = {
			text = jfp_tt_reform_cooldown
			NOT = { has_variable = jfp_cd_manchuria_dev }
		}
	}
	when_taken = {
		set_variable = jfp_dec_manchuria_migration_taken
		set_variable = { name = jfp_cd_manchuria_dev days = 365 }
		jfp_pay_medium = yes
		custom_tooltip = jfp_tt_manchuria_states
		every_state_in_manchuria_old = {
			limit = { owner = root }
			add_modifier = { name = jfp_manchuria_migration days = very_long_modifier_time }
		}
		custom_tooltip = {
			text = jfp_tt_manchuria_dev_up
			change_variable = { name = jfp_manchuria_dev add = 25 }
			clamp_variable = { name = jfp_manchuria_dev min = 0 max = 100 }
		}
	}
	ai_chance = { value = 10 }
}
