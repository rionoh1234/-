# 1. 세도정치
#################################
je_jfp_sedo_politics = {
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	group = je_group_historical_content

	immediate = {
		if = {
			limit = { NOT = { has_variable = jfp_royal_authority } }
			set_variable = { name = jfp_royal_authority value = 0 }
		}
	}

	modifiers_while_active = {
		jfp_sedo_dominance
	}

	current_value = {
		value = 0
		if = {
			limit = { has_variable = jfp_royal_authority }
			value = var:jfp_royal_authority
		}
	}
	goal_add_value = {
		add = 100
	}
	progressbar = yes

	complete = {
		scope:journal_entry = { is_goal_complete = yes }
	}
	on_complete = {
		trigger_event = { id = jfp_sedo.10 popup = yes }
	}

	# 약 28년 (1836 → 1864, 고종 즉위 무렵)
	timeout = 10220
	on_timeout = {
		trigger_event = { id = jfp_sedo.11 popup = yes }
	}

	invalid = {
		NOT = { jfp_is_joseon = yes }
	}

	on_monthly_pulse = {
		random_events = {
			300 = 0
			6 = jfp_sedo.1
			6 = jfp_sedo.2
			5 = jfp_sedo.3
			5 = jfp_sedo.4
			3 = jfp_sedo.5
			4 = jfp_sedo.6
		}
	}

	can_revolution_inherit = yes
	should_be_pinned_by_default_uninvolved_or_context = yes
	weight = 100
}

je_jfp_seohak = {
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	group = je_group_historical_content

	immediate = {
		if = {
			limit = { NOT = { has_variable = jfp_seohak_stance } }
			set_variable = { name = jfp_seohak_stance value = 0 }
		}
	}

	complete = {
		OR = {
			custom_tooltip = {
				text = jfp_tt_seohak_resolved_persecute
				has_variable = jfp_seohak_stance
				var:jfp_seohak_stance >= 3
			}
			custom_tooltip = {
				text = jfp_tt_seohak_resolved_tolerate
				has_variable = jfp_seohak_stance
				var:jfp_seohak_stance <= -3
			}
			has_law_or_variant = law_type:law_freedom_of_conscience
		}
	}
	on_complete = {
		trigger_event = { id = jfp_seohak.9 popup = yes }
	}

	# 약 34년 (~1870)

		}
	}

}
