je_jfp_ryukyu = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	group = je_group_foreign_affairs

	modifiers_while_active = {
		jfp_ryukyu_question
	}

	complete = {
		OR = {
			c:RYU ?= { is_subject_of = root }
			country_or_subject_owns_entire_state_region = STATE_RYUKYU_ISLANDS
		}
	}
	on_complete = {
		trigger_event = { id = jfp_exp.2 popup = yes }
	}

	fail = {
		NOT = { exists = c:RYU }
		NOT = { country_or_subject_owns_entire_state_region = STATE_RYUKYU_ISLANDS }
	}
	on_fail = {
		trigger_event = { id = jfp_exp.3 popup = yes }
	}

	invalid = {
		NOT = { jfp_is_joseon = yes }
	}

	can_revolution_inherit = yes
	should_be_pinned_by_default_uninvolved_or_context = yes
	weight = 150
}
