# 2~5단계 도중 자주권을 잃으면 되찾을 때까지 그 단계를 완료할 수 없습니다.

#################################
# 1. 자주독립
#################################
je_jfp_daeup_1 = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"
	group = je_group_historical_content

	modifiers_while_active = {
		jfp_trial_qing_suzerainty
	}

	complete = {
		is_subject = no
	}
	on_complete = {
		trigger_event = { id = jfp_daeup.1 popup = yes }
	}

	invalid = {
		NOT = { jfp_is_joseon = yes }
	}

	on_monthly_pulse = {
		random_events = {
			200 = 0
			5 = jfp_daeup.101
			5 = jfp_daeup.102
			5 = jfp_daeup.103
		}
	}

	can_revolution_inherit = yes
	should_be_pinned_by_default_uninvolved_or_context = yes
	weight = 200
}

#################################
# 2. 부국강병
#################################
je_jfp_daeup_2 = {
	icon = "gfx/interface/icons/event_icons/event_industry.dds"
	group = je_group_historical_content

	modifiers_while_active = {
		jfp_trial_foreign_creditors
	}

	complete = {
		custom_tooltip = {
			text = jfp_tt_daeup_sovereign
			NOT = { jfp_joseon_lost_sovereignty = yes }
		}
		is_country_type = recognized
		country_rank >= rank_value:major_power
		has_technology_researched = railways
		has_technology_researched = nationalism
		has_technology_researched = general_staff
		custom_tooltip = {
			text = jfp_tt_daeup_2_sectors
			has_variable = jfp_bugguk_sectors
			var:jfp_bugguk_sectors >= 2
		}
		custom_tooltip = {
			text = jfp_tt_daeup_2_japan
			OR = {
				NOT = { exists = c:JAP }
				c:JAP ?= {
					gdp < root.gdp
					army_power_projection < root.army_power_projection
				}
			}
		}
	}
	on_complete = {
		trigger_event = { id = jfp_daeup.2 popup = yes }
	}

	invalid = {
		NOT = { jfp_is_joseon = yes }
	}

	on_monthly_pulse = {
		random_events = {
			200 = 0
			5 = jfp_daeup.201
			5 = jfp_daeup.202
			5 = jfp_daeup.203
		}
	}

	can_revolution_inherit = yes
	should_be_pinned_by_default_uninvolved_or_context = yes
	weight = 200
}
