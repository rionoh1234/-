jfp_button_muwiyeong = {
	name = jfp_button_muwiyeong
	desc = jfp_button_muwiyeong_desc

	visible = {
		has_journal_entry = je_jfp_bugguk_military
		NOT = { has_variable = jfp_bugguk_muwiyeong_done }
	}

	possible = {
		custom_tooltip = {
			text = jfp_tt_reform_cooldown
			NOT = { has_variable = jfp_cd_bugguk_military }
		}
	}

	effect = {
		set_variable = jfp_bugguk_muwiyeong_done
		set_variable = { name = jfp_cd_bugguk_military days = 365 }
		jfp_pay_medium = yes
		add_modifier = { name = jfp_bugguk_muwiyeong days = never_fire_again_modifier_time }
		jfp_ig_up = { IG = ig_armed_forces }
	}

	ai_chance = {
		value = 10
	}
}
