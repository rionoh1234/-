# 이익집단 지지도 증감 (IG = ig_landowners 등)
jfp_ig_up = {
	ig:$IG$ ?= { add_modifier = { name = jfp_ig_approval_up days = normal_modifier_time is_decaying = yes } }
}
jfp_ig_up_large = {
	ig:$IG$ ?= { add_modifier = { name = jfp_ig_approval_up_large days = normal_modifier_time is_decaying = yes } }
}
jfp_ig_down = {
	ig:$IG$ ?= { add_modifier = { name = jfp_ig_approval_down days = normal_modifier_time is_decaying = yes } }
}
jfp_ig_down_large = {
	ig:$IG$ ?= { add_modifier = { name = jfp_ig_approval_down_large days = normal_modifier_time is_decaying = yes } }
}
jfp_ig_stronger = {
	ig:$IG$ ?= { add_modifier = { name = jfp_ig_strength_up days = normal_modifier_time } }
}
jfp_ig_weaker = {
	ig:$IG$ ?= { add_modifier = { name = jfp_ig_strength_down days = normal_modifier_time } }
}

}

}
