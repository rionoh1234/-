# 조선 고봉밥 - 연대기
# 매월 한 번 조선(ROOT) 스코프에서 실행됩니다.
# 각 사건은 jfp_done_* 변수로 한 번만 발생합니다.

# 게임 시작(또는 중간 적용) 시 1회 초기화
jfp_initialize = {
	if = {
		limit = { NOT = { has_variable = jfp_initialized } }
		set_variable = jfp_initialized
		set_variable = { name = jfp_royal_authority value = 0 }
		set_variable = { name = jfp_samjeong_reform value = 0 }
		set_variable = { name = jfp_seohak_stance value = 0 }
		if = {
			limit = { game_date < 1863.1.1 }
			add_journal_entry = { type = je_jfp_sedo_politics }
			add_journal_entry = { type = je_jfp_samjeong }
		}
		if = {
			limit = { game_date < 1866.1.1 }
			add_journal_entry = { type = je_jfp_seohak }
		}
	}
}

jfp_monthly_timeline = {

	# 대업 캠페인 시작 (게임 시작 또는 중간 적용 시 1회)
	#################################
	if = {
		limit = { NOT = { has_variable = jfp_campaign_initialized } }
		set_variable = jfp_campaign_initialized

		trigger_event = { id = jfp_daeup.10 popup = yes }
	}

			game_date >= 1845.7.1

		}
		set_variable = jfp_done_samarang

			has_journal_entry = je_jfp_seohak
			game_date >= 1846.9.1
		}
		set_variable = jfp_done_kimdaegeon
		trigger_event = { id = jfp_seohak.3 popup = yes }

	#################################

	if = {
		limit = {
			NOT = { has_variable = jfp_done_daewongun }
			game_date >= 1864.1.1

			jfp_joseon_sovereign = yes
			country_has_monarchy_law = yes

		set_variable = jfp_done_daewongun

	}

	if = {
		limit = {
			NOT = { has_variable = jfp_done_byeongin_persecution }
			game_date >= 1866.2.1
			game_date < 1872.1.1
			NOT = { has_law_or_variant = law_type:law_freedom_of_conscience }
		}
		set_variable = jfp_done_byeongin_persecution
		trigger_event = { id = jfp_seohak.4 popup = yes }
	}

	if = {
		limit = {
			NOT = { has_variable = jfp_done_yangyo_start }
			game_date >= 1866.6.1
			game_date < 1876.1.1
			jfp_joseon_sovereign = yes
			has_law_or_variant = law_type:law_isolationism
		}

	#################################
	# 1876.2 개항 (강화도 조약 또는 자주적 개국)
	if = {
		limit = {
			NOT = { has_variable = jfp_done_gaehang }
			jfp_joseon_sovereign = yes
			OR = {
				game_date >= 1876.2.1
				AND = {
					game_date >= 1866.1.1
					NOT = { has_law_or_variant = law_type:law_isolationism }
				}
			}
		}
		set_variable = jfp_done_gaehang
		if = {
			limit = { has_law_or_variant = law_type:law_isolationism }
			trigger_event = { id = jfp_gaehwa.1 popup = yes }
		}
		else = {
			trigger_event = { id = jfp_gaehwa.2 popup = yes }
		}
	}
	# 1880.6 수신사 파견
	if = {
		limit = {
			NOT = { has_variable = jfp_done_susinsa }
			has_journal_entry = je_jfp_gaehwa

	}

	if = {

			NOT = { has_variable = jfp_done_gapsin }
			has_journal_entry = je_jfp_gaehwa

	if = {

			game_date >= 1894.4.1
			game_date < 1910.1.1
