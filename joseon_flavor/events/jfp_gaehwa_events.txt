# 강화도 조약 (1876, 쇄국 유지 상태에서)
jfp_gaehwa.1 = {
	type = country_event
	placement = root
	title = jfp_gaehwa.1.t
	desc = jfp_gaehwa.1.d
	flavor = jfp_gaehwa.1.f
	event_image = { video = "unspecific_signed_contract" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_trade.dds"
	duration = 3

	option = {
		name = jfp_gaehwa.1.a
		default_option = yes
		activate_law = law_type:law_mercantilism
		add_modifier = { name = jfp_unequal_treaty days = normal_modifier_time }
		add_modifier = { name = jfp_opened_ports days = normal_modifier_time }
		add_journal_entry = { type = je_jfp_gaehwa }
		if = {
			limit = { exists = c:JAP }
			change_relations = { country = c:JAP value = 20 }
		}
	}
	option = {
		name = jfp_gaehwa.1.b
		add_modifier = { name = jfp_wijeong_cheoksa days = normal_modifier_time }
		jfp_ig_up_large = { IG = ig_devout }
		add_journal_entry = { type = je_jfp_gaehwa }
		if = {
			limit = { exists = c:JAP }
			change_relations = { country = c:JAP value = -30 }
		}
	}
}

			has_variable = jfp_gaehwa

		random_list = {
			40 = {
				add_modifier = { name = jfp_gapsin_success days = normal_modifier_time }
				jfp_gaehwa_up_large = yes
				create_character = {
					first_name = jfp_name_okgyun
					last_name = jfp_name_kim
					birth_date = 1851.2.23
					ig_leader = yes
					interest_group = ig_intelligentsia
					ideology = ideology_reformer
					traits = { ambitious reckless }
				}
			}
			60 = {
				add_modifier = { name = jfp_gapsin_failed days = normal_modifier_time }
				jfp_gaehwa_down = yes
			}
		}
	}
	option = {
		name = jfp_gaehwa.6.b

	}
