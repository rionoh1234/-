# 흥선대원군 집권 (1864)
jfp_dwg.1 = {
	type = country_event
	placement = root
	title = jfp_dwg.1.t
	desc = jfp_dwg.1.d
	flavor = jfp_dwg.1.f
	event_image = { video = "unspecific_ruler_speaking_to_people" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	duration = 3

	option = {
		name = jfp_dwg.1.a
		default_option = yes
		create_character = {
			first_name = jfp_name_haeung
			last_name = jfp_name_yi
			birth_date = 1820.12.21
			ig_leader = yes
			interest_group = ig_landowners
			ideology = ideology_royalist
			traits = { imperious ambitious }
		}
		add_journal_entry = { type = je_jfp_daewongun }
		if = {
			limit = { has_modifier = jfp_royal_authority_restored }
			set_variable = { name = jfp_daewongun_reform value = 10 }
		}
	}
	option = {
		name = jfp_dwg.1.b
		jfp_ig_up = { IG = ig_landowners }
	}
}

	}
}
