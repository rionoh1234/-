jfp_bugguk.1 = {
	type = country_event
	placement = root
	title = jfp_bugguk.1.t
	desc = jfp_bugguk.1.d
	flavor = jfp_bugguk.1.f
	event_image = { video = "unspecific_military_parade" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 3

	option = {
		name = jfp_bugguk.1.a
		default_option = yes
		add_modifier = { name = jfp_bugguk_military_done days = never_fire_again_modifier_time }
		custom_tooltip = jfp_tt_bugguk_sector_count
		change_variable = { name = jfp_bugguk_sectors add = 1 }
		custom_tooltip = jfp_tt_bugguk_recognition
		je:je_earn_recognition ?= {
			add_progress = { value = 40 name = earn_recognition_progress_bar }
		}
	}
}
