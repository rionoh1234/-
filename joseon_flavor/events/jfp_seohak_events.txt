# 저널 종결
jfp_seohak.9 = {
	type = country_event
	placement = root
	title = jfp_seohak.9.t
	desc = jfp_seohak.9.d
	flavor = jfp_seohak.9.f
	event_image = { video = "asia_confucianism_shinto" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	duration = 3

	option = {
		name = jfp_seohak.9.a
		default_option = yes
		trigger = {
			NOT = { has_law_or_variant = law_type:law_freedom_of_conscience }
			OR = {
				NOT = { has_variable = jfp_seohak_stance }
				var:jfp_seohak_stance >= 0
			}
		}
		add_modifier = { name = jfp_orthodoxy_upheld days = long_modifier_time }
	}
	option = {
		name = jfp_seohak.9.b
		trigger = {
			OR = {
				has_law_or_variant = law_type:law_freedom_of_conscience
				AND = {
					has_variable = jfp_seohak_stance
					var:jfp_seohak_stance < 0
				}
			}
		}
		add_modifier = { name = jfp_western_learning days = long_modifier_time }
	}
}
